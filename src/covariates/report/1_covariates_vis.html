Code 
for(cov in sort(unique(cov_df$covariate), decreasing = T)){
  cat(paste0("\n\n## ", cov, "\n\n"))
  
  for(m in unique(master_index$macroregion)){
    # m= 'East'
    g <- ggplot(cov_df |> filter(covariate==cov&macroregion==m) , aes(x = t, y = value_std, color = scale_label)) +
      geom_line() +
      facet_wrap(~i_name, scales = "free_y") +
      labs(
        title = paste(cov, '-' ,m ),
        x = "Time",
        y = "Standardised value",
        color = "(Sum_stat, Time, Spatial)"
      ) +
      theme_minimal()
    print(g)
  }
}
