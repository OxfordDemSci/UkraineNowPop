This report illustrates the correlation between covariates. Because there are 80 covariates, it splits the correlation analysis into three parts:

  1. Correlation between processing methods for each covariate

  2. Correlation between covariates for each processing method

  3. Full correlation table for further exploration.

The correlation is calculated using the Pearson method between covariate values computed for each oblast and time point.

for(cov in sort(unique(cov_df$covariate), decreasing = T)){
  
  cat(paste0("\n\n## ", cov, "\n\n"))
  
  cov_corr_plot <- cov_df |> 
  filter(covariate == cov) |> 
  select(i, t, cov_process, value_std) |> 
  pivot_wider(names_from = cov_process, values_from = value_std) |> 
  select(-i, -t) %>%
  cor(use = "pairwise.complete.obs") |> 
  as.matrix() |> 
    corrplot::corrplot(method = "color", type = "upper", tl.col = "black", tl.srt = 45)
  
  cat("\n")
}

cov_df_corr <- cov_df %>%
  select(i, t, cov_process, value_std) |> 
  pivot_wider(names_from = cov_process, values_from = value_std) |> 
  select(-i, -t) %>%
  cor(use = "pairwise.complete.obs") |> 
  as.data.frame() |> 
  rownames_to_column(var = "cov_process") |> 
  pivot_longer(-cov_process, names_to = "cov_process2", values_to = "correlation") |>
  filter(correlation <0.99) |> 
  rowwise() |> 
  mutate(correlation = round(correlation, 2),
         type = factor(case_when(
           str_split(cov_process, ",")[[1]][1] == str_split(cov_process2, ",")[[1]][1] ~ "same covariate",
           grepl('acled', str_split(cov_process, ",")[[1]][1]) & grepl('acled', str_split(cov_process2, ",")[[1]][1]) ~ "same covariate",
           str_split_fixed(cov_process, ",", 2)[[2]] == str_split_fixed(cov_process2, ",", 2)[[2]] ~ "same processing",
           T ~ "different processing and covariate"
         ))
         ) 

cov_df_corr |> 
  select(type, cov_process, cov_process2, correlation) |>
  DT::datatable(rownames = FALSE, options = list(pageLength = 10, scrollX = TRUE), filter="top")
